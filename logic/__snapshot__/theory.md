# A separation logic for WasmCert

What package `logic` defines and proves, in order, with the kernel's statement of each (every lemma is a theorem: loading the package proves it). Generated by `logic/theory_test.mbt`; run `moon test --update` to refresh it.

Types are abbreviated: `Byte`, `W32`, `W64` are words of 8, 32, 64 bits; `Heap` is `(Num) -> Option[Byte]` (what each address holds); `HProp` is a heap assertion `(Heap) -> Bool`; `Mem` is a WasmCert memory `(Limit_t, List[Byte])`; `Assn` is `(Wstore, Wframe, List[Value]) -> Bool` (an assertion on store, frame and value stack); `SAssn` is `(List[Value], List[Value], List[Value], Heap) -> Bool` (on the value stack, the locals, the globals and the memory as a heap).

## Heaps and separating conjunction (heap.mbt)

definition `hdisj`

```
hdisj((h1 : Heap), (h2 : Heap)) == (∀ x : Num, h1(x) == NONE || h2(x) == NONE)
```

definition `hunion`

```
hunion((h1 : Heap), (h2 : Heap)) == ((x : Num) => if h1(x) == NONE { h2(x) } else { h1(x) })
```

definition `star`

```
star((p : HProp), (q : HProp)) == ((h : Heap) => ∃ h1 : Heap, ∃ h2 : Heap, hdisj(h1, h2) && h == hunion(h1, h2) && p(h1) && q(h2))
```

definition `emp`

```
emp == ((h : Heap) => ∀ x : Num, h(x) == NONE)
```

definition `pts_heap`

```
pts_heap((a : Num), (bs : List[Byte])) == ((x : Num) => if a <= x && x < a + LENGTH(bs) { SOME(EL(x - a, bs)) } else { NONE })
```

definition `pts`

```
pts((a : Num), (bs : List[Byte])) == ((h : Heap) => h == pts_heap(a, bs))
```

definition `heap_of`

```
heap_of((m : Mem)) == ((x : Num) => if x < LENGTH(SND(m)) { SOME(EL(x, SND(m))) } else { NONE })
```

lemma `hdisj_sym`

```
∀ h1 : Heap, ∀ h2 : Heap, hdisj(h1, h2) == hdisj(h2, h1)
```

lemma `hunion_none`

```
∀ h1 : Heap, ∀ h2 : Heap, ∀ x : Num, (hunion(h1, h2, x) == NONE) == (h1(x) == NONE && h2(x) == NONE)
```

lemma `hunion_comm`

```
∀ h1 : Heap, ∀ h2 : Heap, hdisj(h1, h2) → hunion(h1, h2) == hunion(h2, h1)
```

lemma `hdisj_union`

```
∀ h1 : Heap, ∀ h2 : Heap, ∀ h3 : Heap, hdisj(hunion(h1, h2), h3) == (hdisj(h1, h3) && hdisj(h2, h3))
```

lemma `hdisj_union_r`

```
∀ h1 : Heap, ∀ h2 : Heap, ∀ h3 : Heap, hdisj(h1, hunion(h2, h3)) == (hdisj(h1, h2) && hdisj(h1, h3))
```

lemma `hunion_assoc`

```
∀ h1 : Heap, ∀ h2 : Heap, ∀ h3 : Heap, hunion(hunion(h1, h2), h3) == hunion(h1, hunion(h2, h3))
```

lemma `star_comm`

```
∀ p : HProp, ∀ q : HProp, star(p, q) == star(q, p)
```

lemma `star_assoc`

```
∀ p : HProp, ∀ q : HProp, ∀ r : HProp, star(star(p, q), r) == star(p, star(q, r))
```

lemma `star_mono`

```
∀ p : HProp, ∀ p2 : HProp, ∀ q : HProp, ∀ q2 : HProp, ∀ h : Heap, (∀ h1 : Heap, p(h1) → p2(h1)) && (∀ h2 : Heap, q(h2) → q2(h2)) && star(p, q, h) → star(p2, q2, h)
```

## Loads and stores against points-to (memory.mbt)

lemma `length_take`

```
∀ l : List[A], ∀ n : Num, n <= LENGTH(l) → LENGTH(take(n, l)) == n
```

lemma `el_take`

```
∀ l : List[A], ∀ n : Num, ∀ i : Num, i < n && n <= LENGTH(l) → EL(i, take(n, l)) == EL(i, l)
```

lemma `length_drop`

```
∀ l : List[A], ∀ n : Num, LENGTH(drop(n, l)) == LENGTH(l) - n
```

lemma `el_drop`

```
∀ l : List[A], ∀ n : Num, ∀ i : Num, n + i < LENGTH(l) → EL(i, drop(n, l)) == EL(n + i, l)
```

lemma `read_bytes_length`

```
∀ m : Mem, ∀ a : Num, ∀ n : Num, a + n <= LENGTH(SND(m)) → LENGTH(read_bytes(m, a, n)) == n
```

lemma `read_bytes_el`

```
∀ m : Mem, ∀ a : Num, ∀ n : Num, ∀ i : Num, a + n <= LENGTH(SND(m)) && i < n → EL(i, read_bytes(m, a, n)) == EL(a + i, SND(m))
```

lemma `pts_at`

```
∀ a : Num, ∀ bs : List[Byte], ∀ r : HProp, ∀ h : Heap, ∀ x : Num, star(pts(a, bs), r, h) && a <= x && x < a + LENGTH(bs) → h(x) == SOME(EL(x - a, bs))
```

lemma `read_pts`

```
∀ a : Num, ∀ bs : List[Byte], ∀ r : HProp, ∀ m : Mem, star(pts(a, bs), r, heap_of(m)) && !(bs == NIL) → a + LENGTH(bs) <= LENGTH(SND(m)) && read_bytes(m, a, LENGTH(bs)) == bs
```

lemma `length_write`

```
∀ l : List[A], ∀ a : Num, ∀ bs : List[A], a + LENGTH(bs) <= LENGTH(l) → LENGTH(APPEND(take(a, l), APPEND(bs, drop(a + LENGTH(bs), l)))) == LENGTH(l)
```

lemma `el_write`

```
∀ l : List[A], ∀ a : Num, ∀ bs : List[A], ∀ x : Num, a + LENGTH(bs) <= LENGTH(l) && x < LENGTH(l) → EL(x, APPEND(take(a, l), APPEND(bs, drop(a + LENGTH(bs), l)))) == (if a <= x && x < a + LENGTH(bs) { EL(x - a, bs) } else { EL(x, l) })
```

lemma `heap_of_write`

```
∀ m : Mem, ∀ a : Num, ∀ bs : List[Byte], a + LENGTH(bs) <= LENGTH(SND(m)) → heap_of(write_bytes(m, a, bs)) == ((x : Num) => if a <= x && x < a + LENGTH(bs) { SOME(EL(x - a, bs)) } else { heap_of(m, x) })
```

lemma `pts_heap_none`

```
∀ a : Num, ∀ bs : List[Byte], ∀ x : Num, (pts_heap(a, bs, x) == NONE) == !(a <= x && x < a + LENGTH(bs))
```

lemma `write_pts`

```
∀ a : Num, ∀ bs : List[Byte], ∀ bs2 : List[Byte], ∀ r : HProp, ∀ m : Mem, star(pts(a, bs), r, heap_of(m)) && LENGTH(bs2) == LENGTH(bs) && !(bs == NIL) → star(pts(a, bs2), r, heap_of(write_bytes(m, a, bs2)))
```

lemma `takefill_length`

```
∀ l : List[A], ∀ a : A, takefill(a, LENGTH(l), l) == l
```

lemma `load_pts`

```
∀ m : Mem, ∀ n : Num, ∀ off : Num, ∀ bs : List[Byte], ∀ r : HProp, star(pts(n + off, bs), r, heap_of(m)) && !(bs == NIL) → load(m, n, off, LENGTH(bs)) == SOME(bs)
```

lemma `store_pts`

```
∀ m : Mem, ∀ n : Num, ∀ off : Num, ∀ bs : List[Byte], ∀ bs2 : List[Byte], ∀ r : HProp, star(pts(n + off, bs), r, heap_of(m)) && LENGTH(bs2) == LENGTH(bs) && !(bs == NIL) → store(m, n, off, bs2, LENGTH(bs2)) == SOME(write_bytes(m, n + off, bs2)) && star(pts(n + off, bs2), r, heap_of(write_bytes(m, n + off, bs2)))
```

## Triples (triple.mbt)

definition `safe`

```
safe(0, (s : Wstore), (f : Wframe), (es : List[Einstr]), (q : Assn)) == true && safe(SUC((n : Num)), s, f, es, q) == ((∃ vs : List[Value], es == MAP(v_to_e, vs) && q(s, f, vs)) || (∃ s2 : Wstore, ∃ f2 : Wframe, ∃ es2 : List[Einstr], reduce(s, f, es, s2, f2, es2)) && (∀ s2 : Wstore, ∀ f2 : Wframe, ∀ es2 : List[Einstr], reduce(s, f, es, s2, f2, es2) → safe(n, s2, f2, es2, q)))
```

definition `triple`

```
triple((p : Assn), (bs : List[B_e]), (q : Assn)) == (∀ s : Wstore, ∀ f : Wframe, ∀ vs : List[Value], ∀ n : Num, p(s, f, vs) → safe(n, s, f, APPEND(MAP(v_to_e, vs), MAP(Basic, bs)), q))
```

definition `gvals`

```
gvals((s : Wstore), (f : Wframe)) == MAP((a : Num) => g_val(EL(a, s_globs(s))), inst_globs(f_inst(f)))
```

definition `globs_ok`

```
globs_ok((s : Wstore), (f : Wframe)) == (∀ i : Num, i < LENGTH(inst_globs(f_inst(f))) → EL(i, inst_globs(f_inst(f))) < LENGTH(s_globs(s)) && (∀ k : Num, k < LENGTH(inst_globs(f_inst(f))) && EL(k, inst_globs(f_inst(f))) == EL(i, inst_globs(f_inst(f))) → k == i))
```

definition `lift`

```
lift((p : SAssn), (r : HProp), (s : Wstore), (f : Wframe), (vs : List[Value])) == (∃ j : Num, smem_ind(f_inst(f)) == SOME(j) && j < LENGTH(s_mems(s)) && globs_ok(s, f) && star(p(vs, f_locs(f), gvals(s, f)), r, heap_of(EL(j, s_mems(s)))))
```

definition `sl_triple`

```
sl_triple((p : SAssn), (bs : List[B_e]), (q : SAssn)) == (∀ r : HProp, triple(lift(p, r), bs, lift(q, r)))
```

lemma `safe_vals`

```
∀ n : Num, ∀ s : Wstore, ∀ f : Wframe, ∀ vs : List[Value], ∀ q : Assn, q(s, f, vs) → safe(n, s, f, MAP(v_to_e, vs), q)
```

lemma `safe_mono`

```
∀ n : Num, ∀ s : Wstore, ∀ f : Wframe, ∀ es : List[Einstr], ∀ q : Assn, ∀ q2 : Assn, (∀ s : Wstore, ∀ f : Wframe, ∀ vs : List[Value], q(s, f, vs) → q2(s, f, vs)) && safe(n, s, f, es, q) → safe(n, s, f, es, q2)
```

lemma `triple_conseq`

```
∀ p : Assn, ∀ p2 : Assn, ∀ bs : List[B_e], ∀ q : Assn, ∀ q2 : Assn, (∀ s : Wstore, ∀ f : Wframe, ∀ vs : List[Value], p2(s, f, vs) → p(s, f, vs)) && triple(p, bs, q) && (∀ s : Wstore, ∀ f : Wframe, ∀ vs : List[Value], q(s, f, vs) → q2(s, f, vs)) → triple(p2, bs, q2)
```

lemma `lift_mono`

```
∀ p : SAssn, ∀ p2 : SAssn, ∀ r : HProp, ∀ s : Wstore, ∀ f : Wframe, ∀ vs : List[Value], ∀ vs2 : List[Value], (∀ ls : List[Value], ∀ gs : List[Value], ∀ h : Heap, p(vs, ls, gs, h) → p2(vs2, ls, gs, h)) && lift(p, r, s, f, vs) → lift(p2, r, s, f, vs2)
```

lemma `map_snoc`

```
∀ vs : List[Value], ∀ v : Value, APPEND(MAP(v_to_e, vs), [v_to_e(v)]) == MAP(v_to_e, APPEND(vs, [v]))
```

lemma `sl_conseq`

```
∀ p : SAssn, ∀ p2 : SAssn, ∀ bs : List[B_e], ∀ q : SAssn, ∀ q2 : SAssn, (∀ vs : List[Value], ∀ ls : List[Value], ∀ gs : List[Value], ∀ h : Heap, p2(vs, ls, gs, h) → p(vs, ls, gs, h)) && sl_triple(p, bs, q) && (∀ vs : List[Value], ∀ ls : List[Value], ∀ gs : List[Value], ∀ h : Heap, q(vs, ls, gs, h) → q2(vs, ls, gs, h)) → sl_triple(p2, bs, q2)
```

lemma `lift_frame`

```
∀ p : SAssn, ∀ fr : HProp, ∀ r : HProp, lift((vs : List[Value]) => (ls : List[Value]) => (gs : List[Value]) => star(p(vs, ls, gs), fr), r) == lift(p, star(fr, r))
```

lemma `sl_frame`

```
∀ p : SAssn, ∀ bs : List[B_e], ∀ q : SAssn, ∀ fr : HProp, sl_triple(p, bs, q) → sl_triple((vs : List[Value]) => (ls : List[Value]) => (gs : List[Value]) => star(p(vs, ls, gs), fr), bs, (vs : List[Value]) => (ls : List[Value]) => (gs : List[Value]) => star(q(vs, ls, gs), fr))
```

lemma `sl_const`

```
∀ p : SAssn, ∀ c : V_num, sl_triple(p, [EConstNum(c)], (vs : List[Value]) => (ls : List[Value]) => (gs : List[Value]) => (h : Heap) => ∃ vs0 : List[Value], vs == APPEND(vs0, [V_num(c)]) && p(vs0, ls, gs, h))
```

## Lemmas for the instruction rules (rules.mbt)

lemma `snoc_eq`

```
∀ suf : List[A], ∀ x : A, ∀ l : List[A], (APPEND(suf, [x]) == l) == (!(l == NIL) && LAST(l) == x && suf == BUTLAST(l))
```

lemma `nil_eq_sym`

```
∀ l : List[A], (NIL == l) == (l == NIL)
```

lemma `snoc_eq_sym`

```
∀ suf : List[A], ∀ x : A, ∀ l : List[A], (l == APPEND(suf, [x])) == (!(l == NIL) && LAST(l) == x && suf == BUTLAST(l))
```

lemma `no_trap`

```
∀ suf : List[Einstr], ∀ b : B_e, ∀ lh : Lholed, Lfilled(0, lh, [Trap], APPEND(suf, [Basic(b)])) && ALL(is_val, suf) → false
```

lemma `no_trap_x`

```
∀ suf : List[Einstr], ∀ x : Einstr, ∀ lh : Lholed, Lfilled(0, lh, [Trap], APPEND(suf, [x])) && ALL(is_val, suf) && !(x == Trap) → false
```

lemma `append_len_eq`

```
∀ a : List[A], ∀ b : List[A], ∀ c : List[A], ∀ d : List[A], APPEND(a, b) == APPEND(c, d) && LENGTH(b) == LENGTH(d) → a == c && b == d
```

lemma `star_pure`

```
∀ c : Bool, ∀ p : HProp, ∀ r : HProp, ∀ h : Heap, star((h1 : Heap) => c && p(h1), r, h) == (c && star(p, r, h))
```

lemma `star_ex`

```
∀ q : (A, Heap) -> Bool, ∀ r : HProp, ∀ h : Heap, star((h1 : Heap) => ∃ vs0 : A, q(vs0, h1), r, h) == (∃ vs0 : A, star(q(vs0), r, h))
```

lemma `all_suffix`

```
∀ p : (A) -> Bool, ∀ a : List[A], ∀ b : List[A], ∀ c : List[A], ∀ d : List[A], APPEND(a, b) == APPEND(c, d) && ALL(p, APPEND(a, b)) → ALL(p, d)
```
